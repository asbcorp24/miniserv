 Источник питания PRF7100L

 SOUR:VOLT:LEV:IMM:AMPL 5    //установка напряжения 
 SOUR:VOLT:LEV:IMM:AMPL?     //запрос установки


 SOUR:CURR:LEV:IMM:AMPL 2.22 //установка тока ограничения
 SOUR:CURR:LEV:IMM:AMPL?    //запрос установки

:SOUR:MODE CC CV OFF     //источник тока, напряжения, 
:SOUR:MODE?              //запрос установки


SOUR:CURR:PROT:LEV 5    //ограничение на установку максимального тока пример 5 А
SOUR:CURR:PROT:LEV?     //запрос установки

SOUR:VOLT:PROT:LEV 20   //ограничение на установку максимального напряжения пример 20 В
SOUR:VOLT:PROT:LEV?     //запрос установки



OUTP:STAT:IMM ON/OFF/1/0  //включение выходного напряжения
OUTP:STAT:IMM?           //запрос статуса

OUTPut:PROTection:CLEar   //отключение режима срабатывания защиты, когда включен аларм

OUTPut:PROTection:TRIPped?  //запрос состояния защиты, сработала или нет
